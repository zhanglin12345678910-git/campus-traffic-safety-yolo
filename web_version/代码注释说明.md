```python
class Detection(db.Model):
    """
    检测记录数据库模型
    用途：存储每次检测的完整信息（文件、结果、时间等）
    表名：detections
    """
    __tablename__ = 'detections'
    
    # === 主键 ===
    id = db.Column(db.Integer, primary_key=True)  # 自增主键
    session_id = db.Column(db.String(36), nullable=False, 
                          default=lambda: str(uuid.uuid4()))  # 唯一会话ID
    
    # === 文件信息字段 ===
    original_filename = db.Column(db.String(255), nullable=False)  # 原始文件名
    file_path = db.Column(db.String(500), nullable=False)          # 文件存储路径
    result_path = db.Column(db.String(500), nullable=True)         # 结果图片路径
    file_type = db.Column(db.String(20), nullable=False)           # 文件类型(image/video/camera)
    file_size = db.Column(db.Integer, nullable=False)              # 文件大小(字节)
    
    # === 检测结果字段 ===
    detection_time = db.Column(db.Float, nullable=False)   # 检测耗时(毫秒)
    object_count = db.Column(db.Integer, default=0)        # 检测到的对象数量
    main_class = db.Column(db.String(100), nullable=True)  # 主要类别名称
    confidence = db.Column(db.Float, default=0.0)          # 置信度(0-1)
    
    # === 边界框坐标 ===
    bbox_x1 = db.Column(db.Integer, nullable=True)  # 左上角X坐标
    bbox_y1 = db.Column(db.Integer, nullable=True)  # 左上角Y坐标
    bbox_x2 = db.Column(db.Integer, nullable=True)  # 右下角X坐标
    bbox_y2 = db.Column(db.Integer, nullable=True)  # 右下角Y坐标
    
    # === 时间戳 ===
    created_at = db.Column(db.DateTime, default=datetime.utcnow)  # 记录创建时间
    
    def to_dict(self):
        """
        将模型转换为字典格式（用于JSON序列化）
        返回：包含所有字段的字典
        """
        return {
            'id': self.id,
            'session_id': self.session_id,
            'original_filename': self.original_filename,
            'file_type': self.file_type,
            'file_size': self.file_size,
            'detection_time': self.detection_time,
            'object_count': self.object_count,
            'main_class': self.main_class,
            'confidence': self.confidence,
            'bbox': {
                'x1': self.bbox_x1,
                'y1': self.bbox_y1,
                'x2': self.bbox_x2,
                'y2': self.bbox_y2
            } if self.bbox_x1 is not None else None,  # 如果没有边界框则返回None
            'created_at': self.created_at.isoformat()  # ISO格式时间字符串
        }
```

```python
def load_model():
    """
    加载YOLO11模型到内存
    
    功能：
    1. 检查ultralytics包是否可用
    2. 验证模型文件是否存在
    3. 加载模型到全局变量
    
    返回：
        bool: 加载成功返回True，失败返回False

        - ImportError: ultralytics包不可用

        - Exception: 其他加载错误
    """
    global model
    try:
        # 检查YOLO包是否可用
        if not YOLO_AVAILABLE:
            print("❌ ultralytics 包不可用，无法加载模型")

            print(f"📁 请检查路径是否正确")

def allowed_file(filename):
    """
    检查文件扩展名是否允许
    
    参数：
        filename (str): 文件名
    
    返回：
        bool: 允许返回True，不允许返回False
    
    支持的格式：
        图片：png, jpg, jpeg, gif, bmp
        视频：mp4, avi, mov, mkv
    """
    ALLOWED_EXTENSIONS = {'png', 'jpg', 'jpeg', 'gif', 'bmp', 'mp4', 'avi', 'mov', 'mkv'}
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

def get_file_type(filename):
    """
    根据文件扩展名判断文件类型
    
    参数：
        filename (str): 文件名
    
    返回：
        str: 'image' | 'video' | 'unknown'
    """
    ext = filename.rsplit('.', 1)[1].lower()
    if ext in {'png', 'jpg', 'jpeg', 'gif', 'bmp'}:
        return 'image'
    elif ext in {'mp4', 'avi', 'mov', 'mkv'}:
        return 'video'
    return 'unknown'

def process_detection_result(results):
    """
    处理YOLO检测结果
    
    参数：
        results: YOLO模型预测返回的结果对象
    
    返回：
        tuple: (vis_array, info_dict)
            vis_array: 绘制了检测框的图像（NumPy数组）
            info_dict: 检测信息字典，包含：
                - count: 检测到的对象数量
                - main: 主要类别名称
                - conf: 置信度
                - xyxy: 边界框坐标元组 (x1, y1, x2, y2)
    
    逻辑：
    1. 如果没有检测结果，返回None和空信息
    2. 使用result.plot()绘制检测框
    3. 选择置信度最高的对象作为主要结果
    4. 提取类别名称、置信度、边界框坐标
    """
    if not results or len(results) == 0:
        return None, {
            "count": 0,
            "main": "-",
            "conf": 0,
            "xyxy": (None, None, None, None)
        }
    
    res = results[0]
    vis = res.plot()  # 绘制检测框（返回NumPy数组）
    
    info = {
        "count": len(res.boxes),  # 检测到的对象数量
        "main": "-",
        "conf": 0,
        "xyxy": (None, None, None, None)
    }
    
    if len(res.boxes) > 0:
        # 找到置信度最高的检测框
        confs = res.boxes.conf.cpu().numpy()
        idx = int(np.argmax(confs))  # 最大置信度索引
        cls_id = int(res.boxes.cls[idx])  # 类别ID
        name = res.names.get(cls_id, str(cls_id))  # 类别名称
        x1, y1, x2, y2 = res.boxes.xyxy[idx].cpu().numpy().astype(int)  # 边界框坐标
        
        info.update({
            "main": name,
            "conf": float(confs[idx]),
            "xyxy": (int(x1), int(y1), int(x2), int(y2))
        })
    
    return vis, info

def save_result_image(vis_array, session_id):
    """
    保存检测结果图片到磁盘
    
    参数：
        vis_array: OpenCV图像数组（BGR格式）
        session_id: 会话ID（用于生成唯一文件名）
    
    返回：
        str: 保存的文件路径，失败返回None
    
    文件命名：result_{session_id}_{timestamp}.jpg
    """
    try:
        result_filename = f"result_{session_id}_{int(time.time())}.jpg"
        result_path = os.path.join(app.config['RESULT_FOLDER'], result_filename)
        cv2.imwrite(result_path, vis_array)
        return result_path
    except Exception as e:
        print(f"保存结果图片失败: {e}")
        return None
```

```python
@app.route('/api/detect/image', methods=['POST'])
def detect_image():
    """
    图片检测API
    
    请求：
        - 方法: POST
        - 数据: FormData with 'file' field
        - 文件类型: image (png, jpg, jpeg, gif, bmp)
    
    流程：
    1. 验证文件上传
    2. 检查文件格式
    3. 保存上传文件
    4. 执行YOLO检测（conf=0.6, iou=0.5）
    5. 处理检测结果
    6. 保存到数据库
    7. 返回Base64编码的结果图片
    
    返回：
        JSON: {
            success: true,
            session_id: str,
            detection_time: float,
            results: {
                object_count: int,
                main_class: str,
                confidence: float,
                bbox: {x1, y1, x2, y2}
            },
            result_image: str (base64)
        }
    
    错误：
        400: 文件上传错误
        500: 模型未加载或检测失败
    
    优化：
    - 置信度阈值0.6（减少误检）
    - IoU阈值0.5（NMS去重）
    - 图像尺寸640（平衡速度和精度）
    """
    try:
        # === 步骤1: 文件验证 ===
        if 'file' not in request.files:
            return jsonify({'error': '没有上传文件'}), 400
        
        file = request.files['file']
        if file.filename == '':
            return jsonify({'error': '没有选择文件'}), 400
        
        if not allowed_file(file.filename):
            return jsonify({'error': '不支持的文件格式'}), 400
        
        if model is None:
            return jsonify({'error': '模型未加载'}), 500
        
        # === 步骤2: 保存文件 ===
        session_id = str(uuid.uuid4())  # 生成唯一会话ID
        filename = secure_filename(file.filename)  # 安全文件名
        file_path = os.path.join(app.config['UPLOAD_FOLDER'], f"{session_id}_{filename}")
        file.save(file_path)
        
        # === 步骤3: 获取文件信息 ===
        file_size = os.getsize(file_path)
        file_type = get_file_type(filename)
        
        # === 步骤4: 执行YOLO检测 ===
        t0 = time.time()
        # 自动选择设备（GPU优先）
        device = "cuda" if hasattr(model, 'device') and 'cuda' in str(model.device) else "cpu"
        # 🎯 关键参数：conf=0.6（提高置信度减少误检），iou=0.5（NMS去重）
        results = model.predict(file_path, device=device, imgsz=640, conf=0.6, iou=0.5, verbose=False)
        detection_time = (time.time() - t0) * 1000  # 转换为毫秒
        
        # === 步骤5: 处理结果 ===
        vis, info = process_detection_result(results)
        result_path = save_result_image(vis, session_id) if vis is not None else None
        
        # === 步骤6: 保存到数据库 ===
        try:
            detection = Detection(
                session_id=session_id,
                original_filename=filename,
                file_path=file_path,
                result_path=result_path,
                file_type=file_type,
                file_size=file_size,
                detection_time=detection_time,
                object_count=info["count"],
                main_class=info["main"] if info["main"] != "-" else None,
                confidence=info["conf"],
                bbox_x1=info["xyxy"][0],
                bbox_y1=info["xyxy"][1],
                bbox_x2=info["xyxy"][2],
                bbox_y2=info["xyxy"][3]
            )
            db.session.add(detection)
            db.session.commit()
            print(f"✅ 检测记录已保存到数据库: {session_id}")
        except Exception as db_error:
            print(f"❌ 保存到数据库失败: {db_error}")
            db.session.rollback()
            # 继续执行，不影响检测结果返回
        
        # === 步骤7: 编码图片为Base64 ===
        result_image_b64 = None
        if vis is not None:
            _, buffer = cv2.imencode('.jpg', vis)
            result_image_b64 = base64.b64encode(buffer).decode('utf-8')
        
        # === 步骤8: 返回结果 ===
        return jsonify({
            'success': True,
            'session_id': session_id,
            'detection_time': round(detection_time, 2),
            'results': {
                'object_count': info["count"],
                'main_class': info["main"],
                'confidence': round(info["conf"], 3),
                'bbox': {
                    'x1': info["xyxy"][0],
                    'y1': info["xyxy"][1],
                    'x2': info["xyxy"][2],
                    'y2': info["xyxy"][3]
                } if info["xyxy"][0] is not None else None
            },
            'result_image': result_image_b64
        })
    except Exception as e:
        return jsonify({'error': f'检测失败: {str(e)}'}), 500
```

```python
# === 全局视频处理状态 ===
video_processor = None      # 当前视频处理器实例
video_processing = False    # 视频处理状态标志

class VideoProcessor:
    """
    视频处理器类 - 实现实时视频检测和显示
    
    功能：
    1. 在后台线程中逐帧处理视频
    2. 支持抽帧处理（step=2）
    3. 线程安全的帧信息存储
    4. 支持启动/停止控制
    
    属性：
        model: YOLO模型实例
        video_path: 视频文件路径
        session_id: 会话ID
        cap: OpenCV VideoCapture对象
        running: 运行标志
        thread: 处理线程
        current_frame_info: 当前帧信息（线程安全）
        frame_lock: 线程锁
    """
    
    def __init__(self, model, video_path, session_id):
        """初始化视频处理器"""
        self.model = model
        self.video_path = video_path
        self.session_id = session_id
        self.cap = None
        self.running = False
        self.thread = None
        self.current_frame_info = None
        self.frame_lock = threading.Lock()  # 线程锁保护共享数据
    
    def start(self):
        """
        启动视频处理
        
        返回：
            tuple: (success: bool, message: str)
        
        流程：
        1. 打开视频文件
        2. 设置运行标志
        3. 启动后台处理线程
        """
        try:
            self.cap = cv2.VideoCapture(self.video_path)
            if not self.cap.isOpened():
                return False, "无法打开视频文件"
            
            self.running = True
            self.thread = threading.Thread(target=self._process_video, daemon=True)
            self.thread.start()
            
            return True, "视频处理已启动"
        except Exception as e:
            return False, f"启动失败: {str(e)}"
    
    def stop(self):
        """
        停止视频处理
        
        流程：
        1. 设置停止标志
        2. 等待线程结束（最多3秒）
        3. 释放视频资源
        """
        self.running = False
        if self.thread:
            self.thread.join(timeout=3)
        if self.cap:
            self.cap.release()
            self.cap = None
    
    def _process_video(self):
        """
        视频处理主循环（后台线程）
        
        流程：
        1. 逐帧读取视频
        2. 抽帧处理（每2帧处理1帧）
        3. YOLO检测
        4. 编码为Base64
        5. 更新当前帧信息（线程安全）
        6. 控制处理速度（50ms延迟）
        
        优化：
        - 抽帧减少计算量
        - JPEG质量80（减少传输量）
        - conf=0.6（减少误检）
        """
        frame_count = 0
        step = 2  # 抽帧步长（处理每第2帧）
        
        while self.running:
            ret, frame = self.cap.read()
            if not ret:
                break  # 视频结束
            
            # 抽帧处理
            if frame_count % step != 0:
                frame_count += 1
                continue
            
            frame_count += 1
            
            # === 执行检测 ===
            t0 = time.time()
            device = "cuda" if hasattr(self.model, 'device') and 'cuda' in str(self.model.device) else "cpu"
            results = self.model.predict(frame, device=device, imgsz=640, conf=0.6, iou=0.5, verbose=False)
            detection_time = (time.time() - t0) * 1000
            
            # === 处理结果 ===
            vis, info = process_detection_result(results)
            
            # === 编码图像 ===
            result_image_b64 = None
            if vis is not None:
                _, buffer = cv2.imencode('.jpg', vis, [cv2.IMWRITE_JPEG_QUALITY, 80])
                result_image_b64 = base64.b64encode(buffer).decode('utf-8')
            
            # === 更新当前帧信息（线程安全）===
            with self.frame_lock:
                self.current_frame_info = {
                    'success': True,
                    'frame_number': frame_count,
                    'detection_time': round(detection_time, 2),
                    'results': {
                        'object_count': info["count"],
                        'main_class': info["main"],
                        'confidence': round(info["conf"], 3),
                        'bbox': {
```

```python
def get_camera_frame():
    """
    获取摄像头检测帧API
    
    功能：
    - 实时读取摄像头画面
    - 执行YOLO检测
    - 返回Base64编码的结果
    
    流程：
    1. 检查摄像头状态
    2. 读取一帧画面
    3. YOLO检测（conf=0.65, imgsz=416）
    4. 处理结果
    5. 图像优化（缩放、压缩）
    6. Base64编码
    7. 返回JSON
    
    返回：
        JSON: {
            success: true,
            detection_time: float,
            results: {
                object_count: int,
                main_class: str,
                confidence: float,
                bbox: {x1, y1, x2, y2}
            },
            result_image: str (base64)
        }
    
    错误：
        400: 摄像头未启动
        500: 模型未加载、连接断开、读取失败
    
    优化：
    - conf=0.65（最高置信度，减少误检如衣服被识别为警告标志）
    - imgsz=416（更快的检测速度）
    - 图像宽度限制640px（减少传输数据）
    - JPEG质量75（平衡质量和速度）
    """
    global camera_active, camera_capture
    
    try:
        # === 前置检查 ===
        if not camera_active:
            return jsonify({'error': '摄像头未启动'}), 400
        
        if model is None:
            return jsonify({'error': '模型未加载'}), 500
        
        if camera_capture is None or not camera_capture.isOpened():
            return jsonify({'error': '摄像头连接已断开'}), 500
        
        # === 读取一帧 ===
        ret, frame = camera_capture.read()
        
        if not ret:
            return jsonify({'error': '无法读取摄像头画面'}), 500
        
        # === 执行YOLO检测 ===
        t0 = time.time()
        device = "cuda" if hasattr(model, 'device') and 'cuda' in str(model.device) else "cpu"
        # 🎯 关键优化：conf=0.65（最高置信度），imgsz=416（更快）
        results = model.predict(frame, device=device, imgsz=416, conf=0.65, iou=0.5, verbose=False)
        detection_time = (time.time() - t0) * 1000
        
        # === 处理结果 ===
        vis, info = process_detection_result(results)
        
        # === 图像优化 ===
        result_image_b64 = None
        if vis is not None:
            # 缩放图像（减少传输量）
            height, width = vis.shape[:2]
            if width > 640:
                scale = 640 / width
                new_width = 640
                new_height = int(height * scale)
                vis = cv2.resize(vis, (new_width, new_height))
            
            # JPEG压缩编码（质量75）
            encode_params = [cv2.IMWRITE_JPEG_QUALITY, 75]
            _, buffer = cv2.imencode('.jpg', vis, encode_params)
            result_image_b64 = base64.b64encode(buffer).decode('utf-8')
        
        # === 返回结果 ===
        return jsonify({
            'success': True,
            'detection_time': round(detection_time, 2),
            'results': {
                'object_count': info["count"],
                'main_class': info["main"],
                'confidence': round(info["conf"], 3),
                'bbox': {
                    'x1': info["xyxy"][0],
                    'y1': info["xyxy"][1],
                    'x2': info["xyxy"][2],
                    'y2': info["xyxy"][3]
                } if info["xyxy"][0] is not None else None
            },
            'result_image': result_image_b64
        })
        
    except Exception as e:
```

```python
            Detection.main_class,
            func.count(Detection.main_class).label('count')
        ).filter(Detection.main_class.isnot(None)).group_by(Detection.main_class).order_by(
            func.count(Detection.main_class).desc()
        ).limit(5).all()
        
        # ========== 每日趋势（最近7天）==========
        daily_stats = []
        for i in range(6, -1, -1):  # 倒序：7天前 -> 今天
            day_start = today_start - timedelta(days=i)
            day_end = day_start + timedelta(days=1)
            day_count = Detection.query.filter(
                Detection.created_at >= day_start,
                Detection.created_at < day_end
            ).count()
            daily_stats.append({
                'date': day_start.strftime('%m-%d'),  # 格式：MM-DD
                'count': day_count
            })
        
        # ========== 最高置信度记录 ==========
        highest_conf_detection = Detection.query.filter(
            Detection.confidence > 0
        ).order_by(Detection.confidence.desc()).first()
        
        highest_confidence = {
            'class': highest_conf_detection.main_class if highest_conf_detection else '-',
            'confidence': round(highest_conf_detection.confidence * 100, 1) if highest_conf_detection else 0
        }
        
        # ========== 返回结果 ==========
        return jsonify({
            'success': True,
            'stats': {
                # 基础数据
                'total_detections': total_detections,
                'today_detections': today_detections,
                'recent_detections': recent_detections,
                
                # 类型分布
                'image_detections': image_detections,
                'video_detections': video_detections,
                'camera_detections': camera_detections,
                
                # 成功率
                'success_detections': success_detections,
                'success_rate': success_rate,
                
                # 排行榜
                'top_classes': [{'class': cls, 'count': count} for cls, count in top_classes],
                'highest_confidence': highest_confidence,
                
                # 趋势数据
                'daily_trend': daily_stats
            }
        })
    except Exception as e:
        return jsonify({'error': f'获取统计信息失败: {str(e)}'}), 500
```
